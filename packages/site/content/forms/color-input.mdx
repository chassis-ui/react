import { CxColorInput, CxFormLabel } from '@chassis-ui/react'

## Overview

`<CxColorInput>` wraps `<input type="color">` with the `.form-input` class, giving a consistently sized and rounded color swatch with no additional markup required.

<Example>
  <CxFormLabel htmlFor="exampleColorInput">Color picker</CxFormLabel>
  <CxColorInput id="exampleColorInput" defaultValue="#0d6efd" />
</Example>

## Disabled

Add the `disabled` boolean attribute to give the input a grayed-out appearance and remove pointer events.

<Example>
  <CxFormLabel htmlFor="exampleColorInputDisabled">Disabled color picker</CxFormLabel>
  <CxColorInput id="exampleColorInputDisabled" defaultValue="#0d6efd" disabled />
</Example>

## As a form field

Use `label`/`help`/`invalidFeedback`/`validFeedback` directly on `<CxColorInput>` to wrap it in a `.form-field` with an associated label and help text.

<Example>
  <CxColorInput
    label="Accent color"
    help="Used for links and primary buttons."
    defaultValue="#0d6efd"
    id="accentColor"
  />
</Example>
