import { FileInput, FormLabel } from '@chassis-ui/react'

## Overview

`<FileInput>` wraps `<input type="file">` with the same `.form-input` styling as `<TextInput>`, including a themed file-selector button.

<Example>
  <FormLabel htmlFor="formFile">Default file input</FormLabel>
  <FileInput id="formFile" />
</Example>

## Multiple files

Add the `multiple` boolean attribute to allow selecting more than one file.

<Example>
  <FormLabel htmlFor="formFileMultiple">Multiple files input</FormLabel>
  <FileInput id="formFileMultiple" multiple />
</Example>

## Disabled

Add the `disabled` boolean attribute to give the input a grayed-out appearance and remove pointer events.

<Example>
  <FormLabel htmlFor="formFileDisabled">Disabled file input</FormLabel>
  <FileInput id="formFileDisabled" disabled />
</Example>

## Sizing

Apply the `size` property (`small` or `large`) to change the size of the input and its file-selector button.

<Example>
  <FormLabel htmlFor="formFileSm">Small file input</FormLabel>
  <FileInput id="formFileSm" size="small" className="mb-medium" />
  <FormLabel htmlFor="formFileLg">Large file input</FormLabel>
  <FileInput id="formFileLg" size="large" />
</Example>
