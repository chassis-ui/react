  CxProgress, CxProgressBar, CxTable, CxRow, CxCol,
} from '@chassis-ui/react'

export const Dashboard = () => {

    { label: 'Total Users', value: '12,540', delta: '+8%', context: 'primary' },

  ]

                  <div>{stat.label}<div className="h4">{stat.value}</div></div>
                  <CxBadge context={stat.context}>{stat.delta}</CxBadge>

              </CxCardBody>
            </CxCard>

      </CxRow>

## Key points

- Stat cards use `<CxBadge>` for delta indicators — the `context` prop drives the color semantically.
- The orders table uses a `render` function to replace raw status strings with `<CxBadge>` components.
- `<CxListGroup flush items={...} />` renders a borderless activity feed from a plain array.
- Progress bars take `context` directly; `value` and `max` work like the native `<progress>` element.
