import { CxProgress, CxProgressBar, CxButton } from '@chassis-ui/react'

## Example

Progress components are built with two HTML elements, some CSS to set the width, and a few attributes. We don't use [the HTML5 `<progress>` element](https://developer.mozilla.org/en-US/docs/Web/HTML/Element/progress), ensuring you can stack progress bars, animate them, and place text labels over them.

## Basic usage

<Example>
  <CxProgress className="mb-3">
    <CxProgressBar value={0} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar value={25} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar value={50} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar value={75} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar value={100} />
  </CxProgress>
</Example>

## Labels

Add labels to your progress bars by placing text within the `<CxProgressBar>`.

<Example>
  <CxProgress className="mb-3">
    <CxProgressBar value={25}>25%</CxProgressBar>
  </CxProgress>
</Example>

## Height

We only set a `height` value on the `<CxProgress>`, so if you change that value the inner `<CxProgressBar>` will automatically resize accordingly.

<Example>
  <CxProgress height={1} className="mb-3">
    <CxProgressBar value={25}></CxProgressBar>
  </CxProgress>
  <CxProgress height={20} className="mb-3">
    <CxProgressBar value={25}></CxProgressBar>
  </CxProgress>
</Example>

## Backgrounds

Use `context` prop to change the appearance of individual progress bars.

<Example>
  <CxProgress className="mb-3">
    <CxProgressBar context="success" value={25} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar context="info" value={50} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar context="warning" value={75} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar context="danger" value={100} />
  </CxProgress>
</Example>

## Multiple bars

Include multiple progress bars in a progress component if you need.

<Example>
  <CxProgress className="mb-3">
    <CxProgressBar value={15} />
    <CxProgressBar context="success" value={30} />
    <CxProgressBar context="info" value={20} />
  </CxProgress>
</Example>

## Striped

Add `variant="striped"` to any `<CxProgressBar>` to apply a stripe via CSS gradient over the progress bar's background context.

<Example>
  <CxProgress className="mb-3">
    <CxProgressBar context="success" variant="striped" value={25} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar context="info" variant="striped" value={50} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar context="warning" variant="striped" value={75} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar context="danger" variant="striped" value={100} />
  </CxProgress>
</Example>

## Animated stripes

The striped gradient can also be animated. Add `animated` property to `<CxProgressBar>` to animate the stripes right to left via CSS3 animations.

<Example>
  <CxProgress className="mb-3">
    <CxProgressBar context="success" variant="striped" animated value={25} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar context="info" variant="striped" animated value={50} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar context="warning" variant="striped" animated value={75} />
  </CxProgress>
  <CxProgress className="mb-3">
    <CxProgressBar context="danger" variant="striped" animated value={100} />
  </CxProgress>
</Example>
