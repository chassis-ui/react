import { 
  CxButton,
  CxForm,
  CxFloatingInput,
  CxTextInput,
  CxFormLabel,
  CxSelect,
  CxTextarea,
  CxCol,
  CxRow 
} from '@chassis-ui/react'

## Example

Wrap a pair of `<CxTextInput>` and `<CxFormLabel>` elements in `CxFloatingInput` to enable floating labels with textual form fields. A `placeholder` is required on each `<CxTextInput>` as our method of CSS-only floating labels uses the `:placeholder-shown` pseudo-element. Also note that the `<CxTextInput>` must come first so we can utilize a sibling selector (e.g., `~`).

<Example>
  <CxFloatingInput className="mb-medium">
    <CxTextInput type="email" id="floatingInput" placeholder="name@example.com" />
    <CxFormLabel htmlFor="floatingInput">Email address</CxFormLabel>
  </CxFloatingInput>
  <CxFloatingInput>
    <CxTextInput type="password" id="floatingPassword" placeholder="Password" />
    <CxFormLabel htmlFor="exampleFormControlTextarea1">Password</CxFormLabel>
  </CxFloatingInput>
</Example>

When there's a `value` already defined, `<CxFormLabel>`s will automatically adjust to their floated position.

<Example>
  <CxFloatingInput>
    <CxTextInput
      type="email"
      id="floatingInputValue"
      placeholder="name@example.com"
      defaultValue="test@example.com"
    />
    <CxFormLabel htmlFor="floatingInputValue">Input with value</CxFormLabel>
  </CxFloatingInput>
</Example>

## Textareas

By default, `<CxTextarea>`s will be the same height as `<CxTextInput>`s.

<Example>
  <CxFloatingInput>
    <CxTextarea
      id="floatingTextarea"
      placeholder="Leave a comment here"
    ></CxTextarea>
    <CxFormLabel htmlFor="floatingTextarea">Comments</CxFormLabel>
  </CxFloatingInput>
</Example>

To set a custom height on your `<CxTextarea>`, do not use the `rows` attribute. Instead, set an explicit `height` (either inline or via custom CSS).

<Example>
  <CxFloatingInput>
    <CxTextarea
      placeholder="Leave a comment here"
      id="floatingTextarea2"
      style={{ height: '100px' }}
    ></CxTextarea>
    <CxFormLabel htmlFor="floatingTextarea2">Comments</CxFormLabel>
  </CxFloatingInput>
</Example>

## Selects

Other than `<CxTextInput>`, floating labels are only available on `<CxSelect>`s. They work in the same way, but unlike `<CxTextInput>`s, they'll always show the `<CxFormLabel>` in its floated state. **Selects with `size` and `multiple` are not supported.**

<Example>
  <CxFloatingInput>
    <CxSelect id="floatingSelect" aria-label="Floating label select example">
      <option>Open this select menu</option>
      <option value="1">One</option>
      <option value="2">Two</option>
      <option value="3">Three</option>
    </CxSelect>
    <CxFormLabel htmlFor="floatingSelect">Works with selects</CxFormLabel>
  </CxFloatingInput>
</Example>

## Layout

When working with the Chassis grid system, be sure to place form elements within column classes.

<Example>
  <CxRow xs={{gutter: 2}}>
    <CxCol md>
      <CxFloatingInput>
        <CxTextInput type="email" id="floatingInputGrid" placeholder="name@example.com" defaultValue="email@example.com" />
        <CxFormLabel htmlFor="floatingInputGrid">Email address</CxFormLabel>
      </CxFloatingInput>
    </CxCol>
    <CxCol md>
      <CxFloatingInput>
        <CxSelect id="floatingSelectGrid" aria-label="Floating label select example">
          <option>Open this select menu</option>
          <option value="1">One</option>
          <option value="2">Two</option>
          <option value="3">Three</option>
        </CxSelect>
        <CxFormLabel htmlFor="floatingSelectGrid">Works with selects</CxFormLabel>
      </CxFloatingInput>
    </CxCol>
  </CxRow>
</Example>
